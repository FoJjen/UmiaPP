using Microsoft.EntityFrameworkCore;
using UmiaPP.Data;
using UmiaPP.Models;

static void Check(bool condition, string message)
{
	if (!condition) throw new Exception(message);
}

var path = Path.Combine(Path.GetTempPath(), $"umia-quote-links-{Guid.NewGuid():N}.db");
try
{
	var repository = new ProjectRepository(() => new AppDbContext(path));
	await using (var db = new AppDbContext(path))
		await SqliteSchemaSynchronizer.SynchronizeAsync(db);
	var project = await repository.CreateAsync(new Project { Name = "Quote checks", ProjectNumber = "TEST" });
	var otherProject = await repository.CreateAsync(new Project { Name = "Other", ProjectNumber = "OTHER" });
	var first = new LubeSakvaraMangd { ProjectId = project.Id, Produktkod = "A", Leverantor = "Original", PrisNto = 100 };
	var second = new LubeSakvaraMangd { ProjectId = project.Id, Produktkod = "B", PrisNto = 200 };
	var foreign = new LubeSakvaraMangd { ProjectId = otherProject.Id };
	await repository.SaveSakvaraSpecificationRowAsync(first);
	await repository.SaveSakvaraSpecificationRowAsync(second);
	await repository.SaveSakvaraSpecificationRowAsync(foreign);
	Check(!first.IsProduktkodChanged && !first.IsLeverantorChanged, "New article must not be marked changed.");
	first.Produktkod = "Replacement";
	first.Leverantor = "New supplier";
	await repository.SaveSakvaraSpecificationRowAsync(first);
	var reloaded = (await repository.GetByIdAsync(project.Id))!.LubeSakvarorMangder.Single(a => a.Id == first.Id);
	Check(reloaded.IsProduktkodChanged && reloaded.IsLeverantorChanged, "Change markers must survive reload.");
	Check(reloaded.OriginalProduktkod == "A" && reloaded.OriginalLeverantor == "Original", "Original values must survive save.");
	reloaded.Produktkod = "A";
	reloaded.Leverantor = "Original";
	Check(!reloaded.IsProduktkodChanged && !reloaded.IsLeverantorChanged, "Restoring original values must clear markers.");
	await repository.SaveSakvaraSpecificationRowAsync(reloaded);

	var quote = new LubeSakvaraOffert { ProjectId = project.Id, Leverantor = "Quote supplier", OffereratPris = 250, FinalAmount = 240 };
	await repository.SavePackageQuoteAsync(quote, new[] { first.Id });
	await repository.SavePackageQuoteAsync(quote, new[] { second.Id });
	var loaded = (await repository.GetByIdAsync(project.Id))!;
	Check(loaded.LubeSakvarorMangder.Count(a => a.PackageQuoteId == quote.Id) == 2, "Adding must preserve existing links.");
	Check(loaded.LubeSakvarorMangder.Where(a => a.PackageQuoteId == quote.Id).Sum(a => a.PrisNto) == 300, "Linked net baseline must include both articles.");

	// A stale editor has no quote membership but must not overwrite the new link.
	first.Anteckningar = "Edited while quote changed";
	await repository.SaveSakvaraSpecificationRowAsync(first);
	loaded = (await repository.GetByIdAsync(project.Id))!;
	Check(loaded.LubeSakvarorMangder.Single(a => a.Id == first.Id).PackageQuoteId == quote.Id, "Row save must not overwrite quote links.");
	try
	{
		await repository.SetArticleQuoteLinksAsync(project.Id, new[] { first.Id, foreign.Id }, null);
		throw new Exception("Cross-project update was accepted.");
	}
	catch (InvalidOperationException) { }
	loaded = (await repository.GetByIdAsync(project.Id))!;
	Check(loaded.LubeSakvarorMangder.Count(a => a.PackageQuoteId == quote.Id) == 2, "Invalid batch must not partially unlink.");

	await repository.SetArticleQuoteLinksAsync(project.Id, new[] { first.Id, second.Id }, null);
	loaded = (await repository.GetByIdAsync(project.Id))!;
	Check(loaded.LubeSakvarorMangder.Count == 2 && loaded.LubeSakvarorMangder.All(a => a.PackageQuoteId == null), "Unlink must preserve article rows.");
	var savedQuote = loaded.LubeSakvarorOfferter.Single(q => q.Id == quote.Id);
	Check(savedQuote.OffereratPris == 250 && savedQuote.FinalAmount == 240, "Link changes must preserve agreed quote prices.");
	Console.WriteLine("Article identity and quote membership checks passed.");
}
finally
{
	Microsoft.Data.Sqlite.SqliteConnection.ClearAllPools();
	foreach (var suffix in new[] { "", "-wal", "-shm" })
		if (File.Exists(path + suffix)) File.Delete(path + suffix);
}
